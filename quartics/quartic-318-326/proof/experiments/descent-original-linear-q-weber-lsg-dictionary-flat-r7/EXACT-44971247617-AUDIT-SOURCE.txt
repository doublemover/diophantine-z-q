module
import all ActualVendorLSGWeierstrassZetaDef318
import all ActualVendorLSGWeierstrassSigmaDef318
import all ActualVendorLSGWeierstrassSigmaQuasiPeriod318
import all ActualVendorLSGWeierstrassZetaQuasiPeriod318
import all ActualVendorLSGWeierstrassRectInv318
import all ActualVendorLSGWeierstrassBasic318
import all ActualVendorLSGWeierstrassLegendre318
import all ActualVendorLSGWeierstrassResidue318
import all ActualVendorLSGWeierstrassSigmaSign318
import all ActualVendorLSGWeierstrassDegree318
import all ActualVendorLSGWeierstrassAddition318
import all ActualVendorLSGWeierstrassDictionary318
import all ActualVendorLSGLemniscateIndepIntegral318
import all ActualVendorLSGLemniscateIndepSymm318
import all ActualVendorLSGLemniscateIndepRealLine318
import all ActualVendorLSGLemniscateIndepMain318
import Lean.Util.CollectAxioms

set_option maxHeartbeats 0
open Lean Elab Command in
run_cmd do
  let env ← getEnv
  let modules : Array Name := #[`ActualVendorLSGWeierstrassZetaDef318,`ActualVendorLSGWeierstrassSigmaDef318,`ActualVendorLSGWeierstrassSigmaQuasiPeriod318,`ActualVendorLSGWeierstrassZetaQuasiPeriod318,`ActualVendorLSGWeierstrassRectInv318,`ActualVendorLSGWeierstrassBasic318,`ActualVendorLSGWeierstrassLegendre318,`ActualVendorLSGWeierstrassResidue318,`ActualVendorLSGWeierstrassSigmaSign318,`ActualVendorLSGWeierstrassDegree318,`ActualVendorLSGWeierstrassAddition318,`ActualVendorLSGWeierstrassDictionary318,`ActualVendorLSGLemniscateIndepIntegral318,`ActualVendorLSGLemniscateIndepSymm318,`ActualVendorLSGLemniscateIndepRealLine318,`ActualVendorLSGLemniscateIndepMain318]
  let mut count : Nat := 0
  let mut found : Nat := 0
  for idx in [:env.header.modules.size] do
    if modules.contains env.header.modules[idx]!.module then
      found := found+1
      for name in env.header.moduleData[idx]!.constNames do
        let axs ← collectAxioms name
        for ax in axs do
          unless ax == ``propext || ax == ``Quot.sound || ax == ``Classical.choice do
            throwError "Unexpected axiom {ax} in {name}"
        count := count+1
  unless found == modules.size do throwError "Missing imported modules: {found}/{modules.size}"
  logInfo m!"EXPERIMENT_AUDIT_PASS declarations={count} modules={found}"
