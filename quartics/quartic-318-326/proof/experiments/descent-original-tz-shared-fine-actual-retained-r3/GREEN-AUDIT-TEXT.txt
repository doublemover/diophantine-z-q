module
import all ActualTZProductResidueDominance318
import all ActualTZHeterogeneousGridRoots318
import all ActualTZExactZeroIntegerGrid318
import all ActualTZDegreeTwoAlgebra318
import all ActualTZProbeExceptionalWheel318
import all ActualTZSharedFineHybridStencil318
import all ActualTZSharedFineHybridProbes318
import all ActualTZSharedFinePrimeMarker318
import all ActualTZSharedFineHybridOffsetCap318
import all ActualTZSharedFineHybridGrid318
import all ActualPointedCyclicPairing318
import all ActualTZOldProductAverage318
import all ActualTZSharedFineHybridOrdered318
import Lean.Util.CollectAxioms

set_option maxHeartbeats 0
open Lean Elab Command in
run_cmd do
  let env ← getEnv
  let modules : Array Name := #[`ActualTZProductResidueDominance318,`ActualTZHeterogeneousGridRoots318,`ActualTZExactZeroIntegerGrid318,`ActualTZDegreeTwoAlgebra318,`ActualTZProbeExceptionalWheel318,`ActualTZSharedFineHybridStencil318,`ActualTZSharedFineHybridProbes318,`ActualTZSharedFinePrimeMarker318,`ActualTZSharedFineHybridOffsetCap318,`ActualTZSharedFineHybridGrid318,`ActualPointedCyclicPairing318,`ActualTZOldProductAverage318,`ActualTZSharedFineHybridOrdered318]
  let mut count : Nat := 0
  let mut found : Nat := 0
  for idx in [:env.header.modules.size] do
    if modules.contains env.header.modules[idx]!.module then
      found := found+1
      for name in env.header.moduleData[idx]!.constNames do
        let axs ← collectAxioms name
        for ax in axs do
          unless ax == ``propext || ax == ``Quot.sound || ax == ``Classical.choice do
            throwError "Unexpected axiom {ax} in {name}"
        count := count+1
  unless found == modules.size do throwError "Missing imported modules: {found}/{modules.size}"
  logInfo m!"EXPERIMENT_AUDIT_PASS declarations={count} modules={found}"
