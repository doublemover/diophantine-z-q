\begin{theorem}[Arithmetic interpretation]\label{thm:o8-conditional}
For every
$x\in\Q$,
\[
 x\in\Z\quad\Longleftrightarrow\quad
 \exists(v_1,\ldots,v_{15671})\in\Q^{15671},\qquad
 F_8(x,v_1,\ldots,v_{15671})=0.
\]
\end{theorem}
